cmake_minimum_required(VERSION 3.25)
project(KosWeather VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
option(BUILD_TESTING "Build automated tests" OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core DBus Gui Network Qml Quick QuickControls2 LinguistTools)
find_package(KF6WindowSystem QUIET)
qt_standard_project_setup(REQUIRES 6.8)
set(QT_QML_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/qml)
include(cmake/KosApplication.cmake)
add_subdirectory(backend)
add_subdirectory(shared/qml)
add_subdirectory(apps/common)
add_subdirectory(apps/weather)
install(FILES LICENSE THIRD_PARTY_NOTICES.md SOURCE_PROVENANCE.md DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/doc/kos-weather)
