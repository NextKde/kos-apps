cmake_minimum_required(VERSION 3.21)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" KOS_APP_RELEASE_VERSION LIMIT_COUNT 1)
project(KosCalculator VERSION ${KOS_APP_RELEASE_VERSION} LANGUAGES CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)
option(BUILD_TESTING "Build application-local tests" OFF)
option(KOS_FLATPAK_BUILD "Use the Flatpak desktop and activation identity" OFF)
if(KOS_FLATPAK_BUILD)
    set(KOS_FLATPAK_APP_ID "org.nextkde.Calculator")
    set(KOS_FLATPAK_RELEASE_DATE "2026-11-03" CACHE STRING "Application release date (YYYY-MM-DD)")
    if(NOT KOS_FLATPAK_RELEASE_DATE MATCHES "^[0-9][0-9][0-9][0-9]-[0-9][0-9]-[0-9][0-9]$")
        message(FATAL_ERROR "KOS_FLATPAK_RELEASE_DATE must use YYYY-MM-DD")
    endif()
endif()
include(CTest)
include(GNUInstallDirs)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core DBus Gui Qml Quick QuickControls2 LinguistTools)
find_package(KF6WindowSystem 6.0 QUIET)
if(BUILD_TESTING)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
endif()
qt_standard_project_setup(REQUIRES 6.5)
qt_policy(SET QTP0001 NEW)
if(Qt6_VERSION VERSION_GREATER_EQUAL 6.8)
    qt_policy(SET QTP0004 NEW)
endif()
set(QT_QML_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qml")
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(KosApplication)
add_subdirectory(shared/qml)
add_subdirectory(apps/common)
add_subdirectory(apps/calculator)
# Kos.Ui is embedded, never installed into the desktop's shared QML tree.
install(FILES LICENSE SOURCE_PROVENANCE.md DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/kos-calculator)
