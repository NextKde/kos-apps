cmake_minimum_required(VERSION 3.25)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" KOS_APP_RELEASE_VERSION LIMIT_COUNT 1)
project(KosWeather VERSION ${KOS_APP_RELEASE_VERSION} LANGUAGES CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
option(BUILD_TESTING "Build automated tests" OFF)
option(KOS_FLATPAK_BUILD "Use sandbox identity and app-owned backend only" OFF)
if(KOS_FLATPAK_BUILD)
    set(KOS_FLATPAK_APP_ID "org.nextkde.Weather")
    set(KOS_FLATPAK_RELEASE_DATE "2026-11-04" CACHE STRING "Release date (YYYY-MM-DD)")
    if(NOT KOS_FLATPAK_RELEASE_DATE MATCHES "^[0-9][0-9][0-9][0-9]-[0-9][0-9]-[0-9][0-9]$")
        message(FATAL_ERROR "KOS_FLATPAK_RELEASE_DATE must use YYYY-MM-DD")
    endif()
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core DBus Gui Network Qml Quick QuickControls2 LinguistTools)
find_package(KF6WindowSystem QUIET)
qt_standard_project_setup(REQUIRES 6.8)
set(QT_QML_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/qml)
include(cmake/KosApplication.cmake)
add_subdirectory(backend)
add_subdirectory(shared/qml)
add_subdirectory(apps/common)
add_subdirectory(apps/weather)
install(FILES LICENSE THIRD_PARTY_NOTICES.md SOURCE_PROVENANCE.md DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/doc/kos-weather)
